Ternlang Quick Reference
// Types
trit           // -1 | 0 | +1
int            // integer
float          // float
trit[]         // 1D array
trittensor<N>  // N-dim tensor

// Trit literals
let a: trit = 1;   // +1 affirm
let b: trit = 0;   // 0  hold
let c: trit = -1;  // -1 reject

// Built-ins
truth()          // +1
hold()           // 0
conflict()       // -1
consensus(a, b)  // logical merge

// Functions
fn f(x: trit, n: int) -> trit {
    return truth();
}

// Match — all 3 arms required
match v {
    -1 => { return conflict(); }
     0 => { return hold();    }
     1 => { return truth();   }
}

// Directives (Tier 2+)
@sparseskip
fn sparse_layer(w: trit[], x: trit[]) -> trit {
    return consensus(w[0], x[0]);
}

// Propagation operator
let r: trit = risky_fn()?;
// returns conflict() if r == -1

// Exit codes
return truth();    // ✓ affirm  (+1)
return hold();     // ~ hold    (0)
return conflict(); // ✗ reject  (-1)

// Print
print("hello");
println(42);
